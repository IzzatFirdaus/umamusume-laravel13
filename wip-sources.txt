<script setup lang="ts">
/*
 * Where every dataset this tool prints comes from (slice 2.2).
 *
 * Every reference view used to end with its own sentence naming a file path and a section number.
 * The Trainer could neither open the file nor find the section, so the sentence told them less than
 * the one fact it had: this table was transcribed rather than fetched, on a date. This page answers
 * that question once for every dataset.
 *
 * **Two lists, because the two age differently.** A fetched dataset's freshness is a recorded
 * timestamp and goes stale the moment the publisher republishes. A transcribed one's is the day the
 * transcribing pass read the document, and the document may have moved since; the column is headed
 * "Read on" rather than "Updated" for that reason.
 *
 * **A dataset with no recorded fetch says so in words.** It does not print a dash and does not print
 * a zero: "No fetch recorded" and "fetched in March" are different facts, and only one is a problem
 * to act on.
 *
 * **The guide is named, not cited.** `the reference guide` is what the Trainer can be told; its file
 * path is a repository detail that belongs in the commit, not on the screen.
 *
 * Laid out as `database-sources.html`: the `div.info` verification band, then the Tiers card, then the
 * Sources ledger, then the two closing cards in a `g-2` grid. Every table is the ported `table.t` and
 * every card is the ported `.card`, so the card is the scroll region a wide ledger needs.
 * HFW-OVERRIDE (data: ledger columns) and (style: no current row) are recorded on the ledger itself;
 * (data: conflicts) is recorded beside the list below.
 */
import AppLayout from '../../layouts/AppLayout.vue';
import PageHeader from '../../components/ui/PageHeader.vue';
import AbsenceValue from '../../components/AbsenceValue.vue';
import DatabaseNav from '../../components/database/DatabaseNav.vue';
import { Head, Link } from '@inertiajs/vue3';
import { computed, ref } from 'vue';

interface FetchedDataset {
    name: string;
    url: string;
    fetched_at: string | null;
    /**
     * N2.11: which provenance store supplied `fetched_at`.
     *
     * The server unions two arms — `data_sources` (the ledger of requests this tool performed) and
     * the reference rows' own inline `source_url`/`fetched_at` pair (ADR-0003 Amendment R3) — and
     * stamps which one dated the row. A ledger date and an inline date are different facts: the
     * first is a request this build recorded, the second is the reference rows' own account of the
     * same fetch. Null is a recorded state — the row is undated and the cell says "No fetch
     * recorded" — so a channel label is never printed beside an absent date, and no date is ever
     * borrowed from a neighbour (N2.11).
     */
    channel?: 'data_sources' | 'inline' | null;
    note: string;
    /**
     * N6.5: the source's own weight class, read from its config entry.
     *
     * Null is a recorded state: a dataset whose config carries no tier is not classified, and the
     * page must not invent one. This is the source's evidence class (official, maintained reference,
     * GameTora export, community observation), not a confidence score for individual facts.
     */
    tier?: string | null;
    /**
     * D-2 / TD-08: the ids of the *open* conflicts whose `affected_sources` names this row.
     *
     * The server resolves `config/reference.php`'s config keys onto the printed row name and stamps
     * the list here, so the page never re-derives the association from claim prose (the 2026-11-21
     * pass recorded that derivation as unverified). Empty is a recorded state - no open disagreement
     * names this source - and never null, because "no recorded disagreement" and "not read" are
     * different facts.
     */
    conflicts?: string[];
}

interface TranscribedDataset {
    name: string;
    url: string;
    anchored_at: string;
    note: string;
    /** See `FetchedDataset.tier`. Transcribed blocks carry the same config key. */
    tier?: string | null;
    /** See `FetchedDataset.conflicts`. A transcribed row is indexed the same way. */
    conflicts?: string[];
}

/**
 * N6.3: one open disagreement, exactly as the server records it.
 *
 * It is a payload, not a constant in this file. The rows live in
 * `config/reference.php` under a `source.anchor`, so `DatabaseController::sources()` transcribes them
 * the same way it transcribes every other anchored block, and this page prints what it is given.
 * Nothing here classifies, resolves or invents a row.
 */
interface ConflictRow {
    id: string;
    question: string;
    claims: string[];
    state: string;
    /** The screens the disagreement is visible on. A route closes the N6.4 loop back to them. */
    affected_routes: string[];
}

const props = defineProps<{
    docName: string;
    fetched: FetchedDataset[];
    transcribed: TranscribedDataset[];
    /** The recorded date the scenario matrix was checked against its sources. */
    matrixVerifiedAt: string;
    /** N6.3: the open disagreements the reference guide's Source Conflict Log still carries. */
    conflicts: ConflictRow[];
}>();

/*
 * The disagreements the reference guide's Source Conflict Log still carries open
 * (`docs/UMAMUSUME_REFERENCE.md` §7, rows 2, 7 and 10). They are printed as conflicts because the
 * app must not pick a side on the Trainer's behalf: the side that is wrong costs a run. Kept here
 * rather than in a table because they are a reading of two published sources, not row data this
 * build owns; a resolved row leaves this list when the log resolves it.
 *
 * The mood race effect used to sit in this list. It is settled: section 1.3's dated resolution
 * (`:1385`, 2026-09-27) reads the client's own Mood Effect panel against the two guide camps and
 * resolves the race-side figure in GameWith's favour, 4% and 2% rather than 10% and 5%. The mock
 * still lists it as open (`database-sources.html:118`), so the ledger printed three conflicts and the
 * wireframe's tile printed four; the reference won and the difference was recorded on the card.
 *
 * The list is four again as of 2026-11-02, and not the same four: the Artisan Cleat Hammer price
 * opened a fourth row (`0174519c`), so the count now agrees with the mock's tile for a different
 * reason than the one above. The mood race row stays settled and stays out.
 */
/**
 * N6.3: the open rows are the server's, so this file stopped carrying its own copy.
 *
 * The constant that used to sit here had drifted from the record it described: it printed a
 * joined paraphrase in one cell, dropped the `id` a fragment link needs, and could not say which
 * screen a disagreement is visible on. `config/reference.php` holds each row as `id`, `question`,
 * `claims[]`, `state` and `affected_routes[]`, and the controller passes that block through.
 *
 * Each claim prints on its own line rather than joined with a semicolon, because the semicolon
 * was doing the work of a boundary and read as prose this app had written. The list is short and
 * the two sides of a disagreement are the whole point, so they stay separable.
 */
const conflictClaims = (row: ConflictRow): string[] =>
    !Array.isArray(row.claims) ? [] : row.claims.filter((claim): claim is string => typeof claim === 'string' && claim !== '');

/**
 * N6.4: a row's affected routes become doors back to the screens where the disagreement shows.
 * The route arrives as a path (`/database/races`), so the label is derived from it rather than
 * stored twice. An empty list renders nothing: a conflict with no affected screen is a real
 * state, not a gap to fill with a guess.
 */
const ROUTE_LABEL: Record<string, string> = {
    '/database/races': 'Races',
    '/database/shop-items': 'Shop items',
    '/database/scenarios': 'Scenarios',
    '/database/events': 'Events',
    '/database/sparks': 'Sparks',
};

const routeLabel = (route: string): string => ROUTE_LABEL[route] ?? route;

/**
 * N6.5: the ledger narrows by the facts every row actually carries.
 *
 * The plan's third suggested filter was a per-source tier, and until ND-14(b) was approved on
 * 2026-11-22 the rows carried none. The controller now reads `source.tier`, so the tier control
 * below builds its options from the rows themselves rather than inventing a mapping; see the
 * `tierScope` note for that history.
 *
 * Kind is the column's own distinction, and read date is a real state the ledger prints in words
 * ("No fetch recorded"), so both are real things to filter on.
 */
type LedgerKind = 'fetched' | 'transcribed';
type LedgerRead = 'all' | 'dated' | 'undated';

const kindScope = ref<LedgerKind | 'all'>('all');
const readScope = ref<LedgerRead>('all');
const tierScope = ref<string>('');
/**
 * D-2 / TD-08: the conflict-involvement facet.
 *
 * `involved` keeps only rows an open conflict names; `clear` keeps only rows none names. The
 * server already resolved the association (`config/reference.php` `affected_sources` -> printed
 * name), so this reads the stamped list rather than re-deriving it from claim prose. `open` is the
 * only state indexed: a conflict ruled resolved drops off every row's list on its own, so the
 * control can never offer "resolved" as a value.
 */
const conflictScope = ref<'all' | 'involved' | 'clear'>('all');

const matchesKind = (kind: LedgerKind): boolean => kindScope.value === 'all' || kindScope.value === kind;

/** The row's open-conflict ids. A payload without the field reads as no recorded disagreement. */
const conflictsOf = (dataset: { conflicts?: string[] }): string[] =>
    !Array.isArray(dataset.conflicts)
        ? []
        : dataset.conflicts.filter((id): id is string => typeof id === 'string' && id !== '');

const matchesConflict = (dataset: { conflicts?: string[] }): boolean => {
    if (conflictScope.value === 'all') {
        return true;
    }

    return conflictScope.value === 'involved' ? conflictsOf(dataset).length > 0 : conflictsOf(dataset).length === 0;
};

/**
 * N6.5: the source's own weight class, now that the payload carries it.
 *
 * ND-14(b) was owner-approved on 2026-11-22, so the earlier refusal to filter on tier no longer
 * applies. The values come from the rows themselves: a dataset whose config carries no tier
 * contributes no option and never matches a tier filter, so the control cannot offer a class the
 * ledger does not have.
 *
 * This is a source evidence class - official, maintained reference, GameTora export, community
 * observation - and not a quality score. The Tiers card above already explains what each class
 * means, so the options print the class letter and the card carries the meaning.
 */
const tierOptions = computed<string[]>(() => {
    const seen = new Set<string>();

    for (const dataset of [...props.fetched, ...props.transcribed]) {
        if (typeof dataset.tier === 'string' && dataset.tier !== '') {
            seen.add(dataset.tier);
        }
    }

    return Array.from(seen).sort();
});

const matchesTier = (tier: string | null | undefined): boolean =>
    tierScope.value === '' || (typeof tier === 'string' && tier === tierScope.value);

/**
 * "Undated" means the row prints no date at all, which only happens on a fetched dataset whose
 * fetch was never recorded; a transcribed row always carries the day it was read. The absence is
 * read from the same field the cell reads, so the filter and the cell cannot disagree.
 */
const matchesRead = (dated: boolean): boolean => {
    if (readScope.value === 'all') {
        return true;
    }

    return readScope.value === 'dated' ? dated : !dated;
};

const visibleFetched = computed<FetchedDataset[]>(() =>
    matchesKind('fetched')
        ? props.fetched.filter(
            (dataset) =>
                matchesRead(dataset.fetched_at !== null) && matchesTier(dataset.tier) && matchesConflict(dataset),
        )
        : [],
);

const visibleTranscribed = computed<TranscribedDataset[]>(() =>
    matchesKind('transcribed')
        ? props.transcribed.filter(
            (dataset) => matchesRead(true) && matchesTier(dataset.tier) && matchesConflict(dataset),
        )
        : [],
);

const visibleRowCount = computed<number>(() => visibleFetched.value.length + visibleTranscribed.value.length);
const ledgerFiltered = computed<boolean>(
    () =>
        kindScope.value !== 'all' ||
        readScope.value !== 'all' ||
        tierScope.value !== '' ||
        conflictScope.value !== 'all',
);

const clearLedgerFilters = (): void => {
    kindScope.value = 'all';
    readScope.value = 'all';
    tierScope.value = '';
    conflictScope.value = 'all';
};

/*
 * N2.11: the arm that dated a fetched row, in the Trainer's words.
 *
 * The server names the store (`data_sources` ledger or `inline` reference rows) so the screen can
 * tell a recorded request from the reference rows' own stamp. The words name the store a Trainer
 * can act on — "the fetch ledger" and "the reference rows" — rather than the raw table name,
 * because the table name is not a fact the ledger page explains elsewhere. A row the server could
 * not date carries no channel, and this is never asked to label it.
 */
const CHANNEL_LABEL: Record<string, string> = {
    data_sources: 'from the fetch ledger',
    inline: 'from the reference rows',
};

function channelLabel(channel: string | null | undefined): string {
    return channel != null ? (CHANNEL_LABEL[channel] ?? channel) : '';
}
